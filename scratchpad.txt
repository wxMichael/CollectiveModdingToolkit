def set_dwm_accent(hwnd: int, attribute: int, state: int, color: str | None = None) -> None:
	accent_policy = AccentPolicy()

	attributes = WindowCompositionAttributes()
	attributes.Attribute = attribute
	attributes.SizeOfData = sizeof(accent_policy)
	attributes.Data = pointer(accent_policy)

	accent_policy.AccentState = state
	if color:
		accent_policy.GradientColor = color

	windll.user32.SetWindowCompositionAttribute(hwnd, pointer(attributes))

class AccentPolicy(Structure):
	_fields_ = (
		("AccentState", DWORD),
		("AccentFlags", DWORD),
		("GradientColor", DWORD),
		("AnimationId", DWORD),
	)


class WindowCompositionAttributes(Structure):
	_fields_ = (
		("Attribute", DWORD),
		("Data", POINTER(AccentPolicy)),
		("SizeOfData", ULONG),
	)
